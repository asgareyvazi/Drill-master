patterns={'or-zero':[], 'or-empty':[], 'falsey-if':[], 'single-row-or-scalar':[], 'broad-except':[], 'pass':[], 'return-none-false':[]}
for folder in ('core','dialogs','tabs','ui'):

  tree=ast.parse(p.read_text()); rel=str(p.relative_to(root))
  for n in ast.walk(tree):
   kind=None
   if isinstance(n,ast.BoolOp) and isinstance(n.op,ast.Or):
    if any(isinstance(v,ast.Constant) and type(v.value) in (int,float) and v.value==0 for v in n.values):kind='or-zero'
    elif any(isinstance(v,ast.Constant) and v.value=='' or isinstance(v,(ast.List,ast.Dict,ast.Tuple)) and not getattr(v,'elts',getattr(v,'keys',[1])) for v in n.values):kind='or-empty'
   elif isinstance(n,ast.If) and isinstance(n.test,ast.UnaryOp) and isinstance(n.test.op,ast.Not):kind='falsey-if'
   elif isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and n.func.attr in ('first','one','one_or_none','scalar'):kind='single-row-or-scalar'
   elif isinstance(n,ast.ExceptHandler) and (n.type is None or any(isinstance(x,ast.Name) and x.id in ('Exception','BaseException') for x in ast.walk(n.type))):kind='broad-except'
   elif isinstance(n,ast.Pass):kind='pass'
   elif isinstance(n,ast.Return) and (n.value is None or isinstance(n.value,ast.Constant) and (n.value.value is None or n.value.value is False)):kind='return-none-false'
   if kind:patterns[kind].append(dict(file=rel,line=n.lineno,column=n.col_offset,status='DETECTED; NOT AUTO-APPROVED'))
