out = Path("/home/user/m30-controls/m29-replayed")
out.mkdir(parents=True, exist_ok=True)
ledger = []
for name, file, before, after, test in cases:
    path = Path(file)
    original = path.read_bytes()
    source = original.decode()
    assert before in source, (name, "mutation no longer applicable")
    digest = hashlib.sha256(original).hexdigest()
    start = time.monotonic()
    try:
        path.write_text(source.replace(before, after, 1))
        result = subprocess.run(
            [".venv/bin/python", "-m", "pytest", "-q", test, "--tb=short"],
            capture_output=True,
            text=True,
            env=dict(
                os.environ,
                QT_QPA_PLATFORM="offscreen",
                QT_STUB_DIR="/home/user/qt-libs",
                LD_LIBRARY_PATH="/home/user/qt-libs",
            ),
            timeout=180,
        )
        (out / f"mutation-{name}.txt").write_text(result.stdout + result.stderr)
    finally:
        path.write_bytes(original)
    restored = hashlib.sha256(path.read_bytes()).hexdigest() == digest
    ledger.append(
        dict(
            control=name,
            file=file,
            test=test,
            mutation_before=before,
            mutation_after=after,
            returncode=result.returncode,
            expected_failure=result.returncode == 1,
            source_restored=restored,
            seconds=round(time.monotonic() - start, 2),
            source_sha256=digest,
        )
    )
    print(name, result.returncode, restored, flush=True)
    (out / "mutation-controls.json").write_text(json.dumps(ledger, indent=2) + "\n")
    assert restored
assert all(row["expected_failure"] for row in ledger)
