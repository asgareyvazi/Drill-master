 ('scope-candidate', 'core/scope_attribution.py', 'or (rep.section_id is None and rep.well_id in invalid_candidate_wells)', 'or False', 'test_scope_rejects_invalid_unassigned_candidates_without_mutation'),
 ('snapshot-alias', 'core/scope_attribution.py', '[asdict(o) for o in self.outcomes]', '[vars(o) for o in self.outcomes]', 'test_attribution_serialization_does_not_alias_outcomes'),

root=Path('/home/user/m30-controls'); results=[]
for label, file, before, after, test in cases:
 p=Path(file); old=p.read_bytes(); assert before in old.decode(),label
 try:
  p.write_text(old.decode().replace(before,after,1))
  r=subprocess.run(['.venv/bin/python','-m','pytest','tests/test_m30_semantic_regressions.py::'+test,'-ra','--tb=short'],capture_output=True,text=True,timeout=120,env=dict(os.environ,QT_QPA_PLATFORM='offscreen',QT_STUB_DIR='/home/user/qt-libs',LD_LIBRARY_PATH='/home/user/qt-libs'))
  (root/(label+'.txt')).write_text(r.stdout+r.stderr)
 finally:p.write_bytes(old)
 restored=hashlib.sha256(p.read_bytes()).hexdigest()==hashlib.sha256(old).hexdigest()
 results.append(dict(control=label,file=file,test=test,exit_code=r.returncode,source_restored=restored,source_sha256=hashlib.sha256(old).hexdigest(),before=before,after=after))
 print(label,r.returncode,restored,flush=True)
 (root/'new-controls.json').write_text(json.dumps(results,indent=2)+'\n')
 assert r.returncode==1 and restored
